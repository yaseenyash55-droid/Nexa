+package com.nexa.social.utils
+
+import android.content.Context
+import android.media.MediaMetadataRetriever
+import android.net.Uri
+import kotlinx.coroutines.Dispatchers
+import kotlinx.coroutines.withContext
+import java.io.File
+
+object MediaFilePreparer {
+    enum class Kind { IMAGE, VIDEO }
+
+    data class PreparedMedia(
+        val file: File,
+        val mimeType: String,
+        val durationMs: Long? = null
+    )
+
+    suspend fun prepare(
+        context: Context,
+        uri: Uri,
+        kind: Kind,
+        maxBytes: Long,
+        maxDurationMs: Long? = null
+    ): Result<PreparedMedia> = withContext(Dispatchers.IO) {
+        runCatching {
+            val resolver = context.contentResolver
+            val header = ByteArray(16)
+            val read = resolver.openInputStream(uri)?.use { it.read(header) }
+                ?: throw IllegalArgumentException("Unable to read selected media")
+            if (read < 4) throw IllegalArgumentException("Selected media is empty or invalid")
+
+            val detected = detectMimeType(header, resolver.getType(uri))
+                ?: throw IllegalArgumentException("Unsupported media format")
+            if (kind == Kind.IMAGE && !detected.startsWith("image/")) {
+                throw IllegalArgumentException("Choose a JPEG, PNG, WebP, or GIF image")
+            }
+            if (kind == Kind.VIDEO && !detected.startsWith("video/")) {
+                throw IllegalArgumentException("Choose an MP4, WebM, or MOV video")
+            }
+
+            val extension = extensionFor(detected)
+            val temporaryFile = File.createTempFile("nexa_upload_", extension, context.cacheDir)
+            try {
+                resolver.openInputStream(uri)?.use { input ->
+                    temporaryFile.outputStream().buffered().use { output -> input.copyTo(output) }
+                } ?: throw IllegalArgumentException("Unable to open selected media")
+
+                if (temporaryFile.length() <= 0L) throw IllegalArgumentException("Selected media is empty")
+                if (temporaryFile.length() > maxBytes) {
+                    throw IllegalArgumentException("Selected media exceeds the upload size limit")
+                }
+
+                val duration = if (kind == Kind.VIDEO) videoDuration(context, uri) else null
+                if (maxDurationMs != null && duration != null && duration > maxDurationMs) {
+                    throw IllegalArgumentException("Bytes videos must be 60 seconds or shorter")
+                }
+
+                PreparedMedia(temporaryFile, detected, duration)
+            } catch (error: Throwable) {
+                temporaryFile.delete()
+                throw error
+            }
+        }
+    }
+
+    internal fun detectMimeType(header: ByteArray, declaredMime: String?): String? {
+        if (header.size >= 3 && header[0] == 0xFF.toByte() && header[1] == 0xD8.toByte() && header[2] == 0xFF.toByte()) {
+            return "image/jpeg"
+        }
+        if (header.size >= 8 && header[0] == 0x89.toByte() && header[1] == 0x50.toByte() && header[2] == 0x4E.toByte() && header[3] == 0x47.toByte()) {
+            return "image/png"
+        }
+        if (header.size >= 12 && String(header, 0, 4, Charsets.US_ASCII) == "RIFF" && String(header, 8, 4, Charsets.US_ASCII) == "WEBP") {
+            return "image/webp"
+        }
+        if (header.size >= 6 && (String(header, 0, 6, Charsets.US_ASCII) == "GIF87a" || String(header, 0, 6, Charsets.US_ASCII) == "GIF89a")) {
+            return "image/gif"
+        }
+        if (header.size >= 8 && String(header, 4, 4, Charsets.US_ASCII) == "ftyp") {
+            return if (declaredMime == "video/quicktime") "video/quicktime" else "video/mp4"
+        }
+        if (header.size >= 4 && header[0] == 0x1A.toByte() && header[1] == 0x45.toByte() && header[2] == 0xDF.toByte() && header[3] == 0xA3.toByte()) {
+            return "video/webm"
+        }
+        return null
+    }
+
+    private fun extensionFor(mimeType: String): String = when (mimeType) {
+        "image/jpeg" -> ".jpg"
+        "image/png" -> ".png"
+        "image/webp" -> ".webp"
+        "image/gif" -> ".gif"
+        "video/webm" -> ".webm"
+        "video/quicktime" -> ".mov"
+        else -> ".mp4"
+    }

+    private fun videoDuration(context: Context, uri: Uri): Long? {

+        } finally {
+            retriever.release()
+        }
+    }

+package com.nexa.social.utils
+
+import com.nexa.social.NexaApiClient
+
+object MediaUrlResolver {
+    fun resolve(value: String?): String? {
+        val mediaUrl = value?.trim()?.takeIf { it.isNotEmpty() } ?: return null
+        if (mediaUrl.startsWith("https://", ignoreCase = true) ||
+            mediaUrl.startsWith("http://", ignoreCase = true)
+        ) {
+            return mediaUrl
+        }
+        if (mediaUrl.startsWith("//")) return "https:$mediaUrl"
+
+        val serverOrigin = NexaApiClient.BASE_URL.removeSuffix("api/").removeSuffix("/")
+        return "$serverOrigin/${mediaUrl.removePrefix("/")}"
+    }
